#define INVALIDATE_GPU (   t)
Value:
if ((t)) { \
switch ((t)->api) { \
case R_API_SDL: \
if ((t)->gpu_handler.sdl_texture.gpu_texture) { \
SDL_DestroyTexture((t)->gpu_handler.sdl_texture.gpu_texture);\
(t)->gpu_handler.sdl_texture.gpu_texture = NULL; \
} \
break; \
case R_API_GL: \
if ((t)->gpu_handler.gl_texture.gl_id) { \
glDeleteTextures(1, &(t)->gpu_handler.gl_texture.gl_id); \
(t)->gpu_handler.gl_texture.gl_id = 0; \
} \
break; \
default: \
break; \
} \
}
